local function getChar()
    local char = LocalPlayer.Character
    if not char then return nil, nil, nil end
    return char,
        char:FindFirstChildOfClass("Humanoid"),
        char:FindFirstChild("HumanoidRootPart")
end

local function onPress(btn, fn)
    local last = 0
    local function guard()
        local t = tick()
        if t - last < 0.2 then return end
        last = t
        fn()
    end
    btn.MouseButton1Click:Connect(guard)
    btn.TouchTap:Connect(guard)
    btn.Activated:Connect(guard)
end

local function attachDrag(target, handle, isLocked)
    local dragging = false
    local moved = false
    local dragStart = nil
    local startPos = nil

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            moved = false
            if isLocked and isLocked() then return end
            dragging = true
            dragStart = input.Position
            startPos = target.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and dragStart and startPos
            and (input.UserInputType == Enum.UserInputType.MouseMovement
                or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            if math.abs(d.X) > 3 or math.abs(d.Y) > 3 then moved = true end
            target.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    return function() return moved end
end

local function makeMiniButton(key, label, toggle, isOn, registerRefresh)
    local btn = Instance.new("TextButton")
    btn.Name = "Mini_" .. key
    btn.Size = UDim2.fromOffset(104, 28)
    MiniCount = MiniCount + 1
    btn.Position = UDim2.new(0, 10, 0, 60 + (MiniCount - 1) * 34)
    btn.BackgroundColor3 = Color3.fromRGB(30, 26, 40)
    btn.BorderSizePixel = 0
    btn.Text = label
    btn.TextColor3 = Color3.fromRGB(240, 235, 250)
    btn.TextSize = 11
    btn.Font = Enum.Font.GothamBold
    btn.AutoButtonColor = false
    btn.Active = true
    btn.ZIndex = 45
    btn.Parent = ScreenGui

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 8)
    c.Parent = btn

    local s = Instance.new("UIStroke")
    s.Color = Accent
    s.Thickness = 1.2
    s.Transparency = 0.25
    s.Parent = btn

    local moved = attachDrag(btn, btn)

    local function refresh()
        if isOn() then
            btn.BackgroundColor3 = Color3.fromRGB(0, 125, 88)
            s.Color = Color3.fromRGB(0, 220, 140)
        else
            btn.BackgroundColor3 = Color3.fromRGB(30, 26, 40)
            s.Color = Accent
        end
    end

    local last = 0
    local function tapped()
        local t = tick()
        if t - last < 0.2 then return end
        last = t
        if moved() then return end
        toggle()
        refresh()
    end

    btn.MouseButton1Click:Connect(tapped)
    btn.TouchTap:Connect(tapped)

    registerRefresh(refresh)
    refresh()

    return btn
end

        if not on then accent.BackgroundColor3 = c end
    end)

        if tick() - lastLauncher < 0.15 then return end
        toggle()
    end)

local ESPFolder = Instance.new("Folder")
ESPFolder.Name = "ZHub_ESP"

local ESPTracker = {}

    Twisteds    = Color3.fromRGB(255, 60, 60),
    Items       = Color3.fromRGB(255, 150, 50),
    MachineIdle = Color3.fromRGB(255, 255, 255),
    MachineDone = Color3.fromRGB(0, 255, 90),
}

local ESPCfg = {
    ScanActors  = 0.35,
    ScanParts   = 1.20,
    MaxDistance = 900,
    FillAlpha   = 0.65,
    BaseWidth   = 190,
    BaseHeight  = 40,
    ItemPatterns    = { "capsule", "research", "tape" },

                        "npc", "npcs", "monster", "enemy", "enemies", "mob" },
}

local PROG_KEYS = { "progress", "completion", "complete", "completed", "done",
                    "finished", "percent", "repaired", "fixed", "filled", "value" }

local function espGet(cat, variant)
    return State[cat .. "ESP_" .. variant] == true
end

local function matchesAny(name, patterns)
    local ln = string.lower(name)
    for _, p in ipairs(patterns) do
        if string.find(ln, p, 1, true) then return true end
    end
    return false
end

local function textScale()
    return State.TextSize / 50
end

local function getPos(inst)
    if inst:IsA("BasePart") then return inst.Position end
    if inst:IsA("Model") then
        local okp, cf = pcall(function() return inst:GetPivot() end)
        if okp and cf then return cf.Position end
        local pp = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
        if pp then return pp.Position end
    end
    return nil
end

local function normalize(v)
    if v <= 0 then return 0 end
    if v <= 1.001 then return math.clamp(math.floor(v * 100 + 0.5), 0, 100) end
    return math.clamp(math.floor(v + 0.5), 0, 100)
end

local function getMachineProgress(inst)
    for _, key in ipairs(PROG_KEYS) do
        local okA, v = pcall(function() return inst:GetAttribute(key) end)
        if okA and type(v) == "number" then return normalize(v) end
    end
    for _, c in ipairs(inst:GetDescendants()) do
        if c:IsA("NumberValue") or c:IsA("IntValue") then
            if matchesAny(c.Name, PROG_KEYS) then
                return normalize(c.Value)
            end
        end
    end
    return nil
end

local function isMachineDone(inst)
    local p = getMachineProgress(inst)
    if p and p >= 100 then return true end
    for _, key in ipairs(PROG_KEYS) do
        local okA, v = pcall(function() return inst:GetAttribute(key) end)
        if okA and type(v) == "boolean" and v then return true end
    end
    for _, c in ipairs(inst:GetDescendants()) do
        if c:IsA("BoolValue") and c.Value and matchesAny(c.Name, PROG_KEYS) then
            return true
        end
    end
    return false
end

local function colorFor(cat, inst)
    if cat == "Machines" then
        return isMachineDone(inst) and ESPCol.MachineDone or ESPCol.MachineIdle
    end
    return ESPCol[cat]
end

local function headOffset(inst)
    if inst:IsA("Model") then
        local okS, size = pcall(function() return inst:GetExtentsSize() end)
        if okS and size then return size.Y / 2 + 2.5 end
    elseif inst:IsA("BasePart") then
        return inst.Size.Y / 2 + 2.5
    end
    return 3.4
end

local function espCleanup(seen, cats)
    for inst, rec in pairs(ESPTracker) do
        if cats[rec.cat] and (not seen[inst] or not inst.Parent) then
            if rec.hl then rec.hl:Destroy() end
            if rec.bb then rec.bb:Destroy() end
            ESPTracker[inst] = nil
        end
    end
end

local function espEnsure(inst, cat, name)
    local wantHL = espGet(cat, "NoText")
    local wantBB = espGet(cat, "Text") and State.TextSize > 0

    if not wantHL and not wantBB then
        local old = ESPTracker[inst]
        if old then
            if old.hl then old.hl:Destroy() end
            if old.bb then old.bb:Destroy() end
            ESPTracker[inst] = nil
        end
        return
    end

    local rec = ESPTracker[inst]
    if not rec then
        rec = { cat = cat }
        ESPTracker[inst] = rec
    end
    rec.cat = cat
    rec.name = name

    local col = colorFor(cat, inst)

    if wantHL then
        if not rec.hl then
            local h = Instance.new("Highlight")
            h.Name = "ZHub_HL"
            h.Adornee = inst
            h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            h.FillTransparency = ESPCfg.FillAlpha
            h.OutlineTransparency = 0.1
            h.Parent = ESPFolder
            rec.hl = h
        end
        rec.hl.FillColor = col
        rec.hl.OutlineColor = col
    elseif rec.hl then
        rec.hl:Destroy()
        rec.hl = nil
    end

    if wantBB then
        if not rec.bb then
            local bb = Instance.new("BillboardGui")
            bb.Name = "ZHub_Tag"
            bb.Adornee = inst
            bb.AlwaysOnTop = true
            bb.LightInfluence = 0
            bb.MaxDistance = ESPCfg.MaxDistance
            bb.Parent = ESPFolder

            local lbl = Instance.new("TextLabel")
            lbl.BackgroundTransparency = 1
            lbl.Size = UDim2.fromScale(1, 1)
            lbl.Font = Enum.Font.GothamBold
            lbl.TextScaled = true
            lbl.TextStrokeTransparency = 0.3
            lbl.TextStrokeColor3 = Color3.new(0, 0, 0)
            lbl.Parent = bb

            rec.bb = bb
            rec.lbl = lbl
        end

        local s = textScale()
        rec.bb.Size = UDim2.fromOffset(ESPCfg.BaseWidth * s, ESPCfg.BaseHeight * s)
        rec.bb.StudsOffsetWorldSpace = Vector3.new(0, headOffset(inst), 0)

        if rec.lbl then
            rec.lbl.TextColor3 = col
            if cat == "Machines" then
                local pct = getMachineProgress(inst)
                local pctTxt
                if isMachineDone(inst) then
                    pctTxt = "100%"
                elseif pct then
                    pctTxt = tostring(pct) .. "%"
                else
                    pctTxt = "?%"
                end
                rec.lbl.Text = name .. "  [" .. pctTxt .. "]"
            else
                rec.lbl.Text = name
            end
        end
    elseif rec.bb then
        rec.bb:Destroy()
        rec.bb = nil
        rec.lbl = nil
    end
end

local function applyTextScale()
    local s = textScale()
    for _, rec in pairs(ESPTracker) do
        if rec.bb then
            rec.bb.Size = UDim2.fromOffset(ESPCfg.BaseWidth * s, ESPCfg.BaseHeight * s)
        end
    end
end

local function isUnderEntityFolder(inst)
    local anc = inst.Parent
    while anc and anc ~= Workspace do
        if matchesAny(anc.Name, ESPCfg.FolderHints) then return true end
        anc = anc.Parent
    end
    return false
end

local function isTwistedModel(m)
    if m == LocalPlayer.Character then return false end
    local underEntity = isUnderEntityFolder(m)
    if Players:GetPlayerFromCharacter(m) and not underEntity then return false end
    if m:FindFirstChildOfClass("Humanoid") then return true end
    if underEntity then
        return (m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")) ~= nil
    end
    return false
end

local function scanActors()
    local seen = {}
    local cam = Workspace.CurrentCamera

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local m = plr.Character
            local hum = m:FindFirstChildOfClass("Humanoid")
            local root = m:FindFirstChild("HumanoidRootPart")
            if hum and root and hum.Health > 0 then
                local dist = cam and (cam.CFrame.Position - root.Position).Magnitude or 0
                if dist <= ESPCfg.MaxDistance then
                    seen[m] = true
                    espEnsure(m, "Players", plr.Name)
                end
            end
        end
    end

    local function visit(container, depth)
        if depth > 5 then return end
        for _, obj in ipairs(container:GetChildren()) do
            if obj ~= ESPFolder then
                if obj:IsA("Folder") then
                    visit(obj, depth + 1)
                elseif obj:IsA("Model") then
                    if isTwistedModel(obj) then
                        seen[obj] = true
                        espEnsure(obj, "Twisteds", obj.Name)
                    elseif not obj:FindFirstChildOfClass("Humanoid") then
                        visit(obj, depth + 1)
                    end
                end
            end
        end
    end
    visit(Workspace, 0)

    espCleanup(seen, { Players = true, Twisteds = true })
end

local function scanParts()
    local seen = {}
    local cam = Workspace.CurrentCamera

    for _, d in ipairs(Workspace:GetDescendants()) do
        local isModel = d:IsA("Model")
        local isPart = d:IsA("BasePart")

        if isModel or isPart then
            local cat = nil
            if matchesAny(d.Name, ESPCfg.ItemPatterns) then
                cat = "Items"
            elseif matchesAny(d.Name, ESPCfg.MachinePatterns) then
                cat = "Machines"
            end

            if cat then
                local skip = false
                if isPart then
                    local anc = d.Parent
                    while anc and anc ~= Workspace do
                        if anc:IsA("Model") and (matchesAny(anc.Name, ESPCfg.ItemPatterns)
                            or matchesAny(anc.Name, ESPCfg.MachinePatterns)) then
                            skip = true
                            break
                        end
                        anc = anc.Parent
                    end
                end

                if not skip then
                    local p = getPos(d)
                    if p and cam and (cam.CFrame.Position - p).Magnitude <= ESPCfg.MaxDistance then
                        seen[d] = true
                        espEnsure(d, cat, d.Name)
                    end
                end
            end
        end
    end

    espCleanup(seen, { Items = true, Machines = true })
end

local function espRefresh()
    task.spawn(function() pcall(scanActors) end)
    task.spawn(function() pcall(scanParts) end)
end

task.spawn(function()
    while ScreenGui.Parent do
        pcall(scanActors)
        task.wait(ESPCfg.ScanActors)
    end
end)

task.spawn(function()
    while ScreenGui.Parent do
        pcall(scanParts)
        task.wait(ESPCfg.ScanParts)
    end
end)

local LightingSaved = nil

local function captureLighting()
    if LightingSaved then return end
    LightingSaved = {
        Brightness = Lighting.Brightness,
        Ambient = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        ClockTime = Lighting.ClockTime,
        FogEnd = Lighting.FogEnd,
        FogStart = Lighting.FogStart,
        GlobalShadows = Lighting.GlobalShadows,
        ShadowSoftness = Lighting.ShadowSoftness,
        EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
        EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
        ExposureCompensation = Lighting.ExposureCompensation,
    }
end

local function restoreLighting()
    if not LightingSaved then return end
    for k, v in pairs(LightingSaved) do
        pcall(function() Lighting[k] = v end)
    end
end

local POST_CLASSES = { "BlurEffect", "BloomEffect", "SunRaysEffect", "DepthOfFieldEffect" }
local PostSaved = {}

local function applyPost()
    if State.AntiLag then
        for _, obj in ipairs(Lighting:GetChildren()) do
            if table.find(POST_CLASSES, obj.ClassName) and obj.Enabled ~= nil then
                if PostSaved[obj] == nil then PostSaved[obj] = obj.Enabled end
                obj.Enabled = false
            end
        end
    else
        for obj, st in pairs(PostSaved) do
            if obj.Parent then obj.Enabled = st end
        end
        table.clear(PostSaved)
    end
end

local function refreshLighting()
    if State.Fullbright or State.AntiLag then captureLighting() end
    restoreLighting()

    if State.Fullbright then
        Lighting.Brightness = 2
        Lighting.Ambient = Color3.fromRGB(125, 125, 125)

local function collectMachines()
    local out = {}
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("Model") and matchesAny(d.Name, ESPCfg.MachinePatterns) then
            local skip = false
            local anc = d.Parent
            while anc and anc ~= Workspace do
                if anc:IsA("Model") and matchesAny(anc.Name, ESPCfg.MachinePatterns) then
                    skip = true
                    break
                end
                anc = anc.Parent
            end
            if not skip then table.insert(out, d) end
        end
    end
    return out
end

local function collectCapsules()
    local out = {}
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("Model") or d:IsA("BasePart") then
            if matchesAny(d.Name, { "capsule", "research" }) then
                local skip = false
                if d:IsA("BasePart") then
                    local anc = d.Parent
                    while anc and anc ~= Workspace do
                        if anc:IsA("Model") and matchesAny(anc.Name, { "capsule", "research" }) then
                            skip = true
                            break
                        end
                        anc = anc.Parent
                    end
                end
                if not skip then table.insert(out, d) end
            end
        end
    end
    return out
end

    TweenService:Create(xCore, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = UDim2.fromOffset(30, 30)
    }):Play()
    TweenService:Create(xGlow, TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = UDim2.fromOffset(130, 130)
    }):Play()

    spinX = true
    xSpin = 60

    task.wait(0.45)

    for i = 1, 44 do
        local a = math.random() * math.pi * 2
        local r = 0.24 + math.random() * 0.36
        local px = 0.5 + math.cos(a) * r
        local py = 0.5 + math.sin(a) * r * 0.9

        local star = makeStar(ov, px, py, math.random(2, 5), 96)
        star.dot.BackgroundTransparency = 1
        star.crossH.BackgroundTransparency = 1
        star.crossV.BackgroundTransparency = 1

        table.insert(twinkleStars, {
            star = star,
            ph = math.random() * 6.28,
            sp = 1.5 + math.random() * 3,
            rot = (math.random() * 40 - 20),
        })

        task.delay(i * 0.02, function()
            TweenService:Create(star.dot, TweenInfo.new(0.4), { BackgroundTransparency = 0.1 }):Play()
            TweenService:Create(star.crossH, TweenInfo.new(0.4), { BackgroundTransparency = 0.3 }):Play()
            TweenService:Create(star.crossV, TweenInfo.new(0.4), { BackgroundTransparency = 0.3 }):Play()
        end)
    end

    task.wait(0.45)

    -- 4) TEXTS
    TweenService:Create(welcome, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        TextTransparency = 0
    }):Play()
    TweenService:Create(brand, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        TextTransparency = 0
    }):Play()
    TweenService:Create(credit, TweenInfo.new(0.4), { TextTransparency = 0 }):Play()

    xSpin = 130
    task.wait(2.1)

    -- 5) TEXTS SHRINK INTO A TINY SPACE AND VANISH (X keeps spinning)
    TweenService:Create(textScaleObj, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
        Scale = 0.02
    }):Play()
    TweenService:Create(welcome, TweenInfo.new(0.4), { TextTransparency = 1 }):Play()
    TweenService:Create(brand, TweenInfo.new(0.4), { TextTransparency = 1 }):Play()
    TweenService:Create(credit, TweenInfo.new(0.4), { TextTransparency = 1 }):Play()

    task.wait(0.5)

    TweenService:Create(beamA, TweenInfo.new(0.6), { Size = UDim2.fromOffset(1020, 18) }):Play()
    TweenService:Create(beamB, TweenInfo.new(0.6), { Size = UDim2.fromOffset(1020, 18) }):Play()
    xSpin = 72

    return {
        overlay = ov,
        cleanup = function()
            dead = true
            suckOn = false
            renderConn:Disconnect()
        end,
    }
end

            b.BorderSizePixel = 0

                BackgroundTransparency = 0,
                TextTransparency = 0
            }):Play()

                        else
                            TweenService:Create(c, TweenInfo.new(0.55), { BackgroundTransparency = 1 }):Play()
                        end
                    end
                end

                TweenService:Create(overlay, TweenInfo.new(0.55), {
                    BackgroundTransparency = 1
                }):Play()

                task.wait(0.6)
                overlay:Destroy()
            end)
        end
    end)
end
